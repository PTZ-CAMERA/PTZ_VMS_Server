cmake_minimum_required(VERSION 3.16)
project(MiniVmsServer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(Threads REQUIRED)
find_package(FFmpeg REQUIRED)
find_package(Boost 1.70 REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(CURL 7.85 REQUIRED)
find_package(tinyxml2 REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(vms_core src/ptz/PtzCommandRouter.cpp src/core/Config.cpp src/core/Logger.cpp src/stream/RtspSession.cpp src/stream/StreamManager.cpp src/client/WebSocketServer.cpp src/stream/RtspRelayServer.cpp src/onvif/CurlOnvifClient.cpp src/camera/CameraService.cpp src/recording/MuxRecorder.cpp src/recording/RecordingManager.cpp src/database/RecordingRepository.cpp)
target_include_directories(vms_core PUBLIC include)
target_compile_definitions(vms_core PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
if(WIN32)
    target_compile_definitions(vms_core PUBLIC _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(vms_core PUBLIC ws2_32 mswsock)
endif()
target_link_libraries(vms_core PUBLIC FFmpeg::FFmpeg Threads::Threads Boost::headers nlohmann_json::nlohmann_json CURL::libcurl tinyxml2::tinyxml2 SQLite::SQLite3)
if(MSVC)
    target_compile_options(vms_core PUBLIC /W4 /permissive- /utf-8 /we4996)
else()
    target_compile_options(vms_core PUBLIC -Wall -Wextra -Wpedantic -Werror=deprecated-declarations)
endif()
add_executable(vms-server src/main.cpp)
target_link_libraries(vms-server PRIVATE vms_core)
configure_file(config/vms.conf config/vms.conf COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_executable(vms-unit-tests tests/unit_tests.cpp)
    target_link_libraries(vms-unit-tests PRIVATE vms_core)
    add_test(NAME vms-unit-tests COMMAND vms-unit-tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME vms-ptz-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/ptz_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264")
        set_tests_properties(vms-ptz-integration PROPERTIES TIMEOUT 45)
        add_test(NAME vms-discovery-registration-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/discovery_registration_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264")
        set_tests_properties(vms-discovery-registration-integration PROPERTIES TIMEOUT 35)
        add_test(NAME vms-rtsp-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/rtsp_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264")
        set_tests_properties(vms-rtsp-integration PROPERTIES TIMEOUT 45)
        add_test(NAME vms-websocket-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/websocket_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264")
        set_tests_properties(vms-websocket-integration PROPERTIES TIMEOUT 45)
        find_program(FFPROBE_EXECUTABLE ffprobe)
        if(FFPROBE_EXECUTABLE)
            add_test(NAME vms-recording-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/recording_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264" "${FFPROBE_EXECUTABLE}")
        else()
            add_test(NAME vms-recording-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/recording_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264")
        endif()
        set_tests_properties(vms-recording-integration PROPERTIES TIMEOUT 75)
        if(FFPROBE_EXECUTABLE)
            add_test(NAME vms-rtsp-udp-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/rtsp_udp_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264" "${FFPROBE_EXECUTABLE}")
        else()
            add_test(NAME vms-rtsp-udp-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/rtsp_udp_integration.py" "$<TARGET_FILE:vms-server>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.h264")
        endif()
        set_tests_properties(vms-rtsp-udp-integration PROPERTIES TIMEOUT 60)
    endif()
endif()
